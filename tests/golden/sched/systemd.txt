=== /srv/hm/.config/systemd/user/meta-controller-tick.service
[Unit]
Description=meta-controller tick (norn status + one check-in per project)

[Service]
Type=oneshot
WorkingDirectory=/srv/meta-controller
ExecStart="/opt/node/bin/node" "/srv/meta-controller/engine/bin/norn.mjs" "tick" "--reason" "schedule"
KillMode=process
TimeoutStartSec=55min
=== /srv/hm/.config/systemd/user/meta-controller-tick.timer
[Unit]
Description=meta-controller tick every 15 min

[Timer]
OnBootSec=2min
OnUnitActiveSec=15min

[Install]
