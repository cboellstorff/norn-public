=== /srv/hm/.config/systemd/user/norn-tick.service
[Unit]
Description=Norn tick (norn status + one check-in per project)

[Service]
Type=oneshot
WorkingDirectory=/srv/norn
ExecStart="/opt/node/bin/node" "/srv/norn/engine/bin/norn.mjs" "tick" "--reason" "schedule"
KillMode=process
TimeoutStartSec=55min
=== /srv/hm/.config/systemd/user/norn-tick.timer
[Unit]
Description=Norn tick every 15 min

[Timer]
OnBootSec=2min
OnUnitActiveSec=15min
AccuracySec=1min

[Install]
WantedBy=timers.target
$ systemctl --user daemon-reload
$ systemctl --user enable --now norn-tick.timer
--- remove
$ systemctl --user disable --now norn-tick.timer
$ rm /srv/hm/.config/systemd/user/norn-tick.service
$ rm /srv/hm/.config/systemd/user/norn-tick.timer
$ systemctl --user daemon-reload
